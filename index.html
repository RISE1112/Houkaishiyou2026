// ============================================================ // 会話練習チャット用 Anthropic API プロキシ (Cloudflare Worker) // ============================================================ // フロントエンド(index.html / rise1112.github.io)から system と // messages だけを受け取り、このWorkerがサーバー側に保管した // ANTHROPIC_API_KEY を付けて api.anthropic.com に転送します。 // これにより、APIキーをブラウザ側のコード(=誰でも見られるGitHubの // リポジトリ)に書かずに済みます。 // // デプロイ手順は README.md を参照してください。 // ============================================================ // あなたのGitHub PagesのURL。ここ以外のサイトからのリクエストは拒否する。 const ALLOWED_ORIGIN = 'https://rise1112.github.io'; const MODEL = 'claude-sonnet-5'; const MAX_TOKENS = 700; // 1つのIPが1日に送れるメッセージ数の上限。青天井の「完全無制限」は // 誰かに乱用された場合に高額請求につながるため、必ず何らかの上限を設けることを推奨します。 const DAILY_LIMIT_PER_IP = 60; // 1リクエストで送られてくる会話履歴の最大メッセージ数(異常に長い会話を防ぐ) const MAX_HISTORY_MESSAGES = 80; function corsHeaders(origin) { const allow = (origin === ALLOWED_ORIGIN) ? origin : ALLOWED_ORIGIN; return { 'Access-Control-Allow-Origin': allow, 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Vary': 'Origin' }; } function jsonError(origin, status, message) { return new Response( JSON.stringify({ type: 'error', error: { type: 'invalid_request', message } }), { status, headers: { ...corsHeaders(origin), 'Content-Type': 'application/json' } } ); } export default { async fetch(request, env) { const origin = request.headers.get('Origin') || ''; if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders(origin) }); } if (request.method !== 'POST') { return jsonError(origin, 405, 'Method not allowed'); } if (origin !== ALLOWED_ORIGIN) { return jsonError(origin, 403, 'このオリジンからのアクセスは許可されていません'); } // ---- レート制限 (IPごとに1日あたりの上限) ---- if (env.RATE_LIMIT) { const ip = request.headers.get('CF-Connecting-IP') || 'unknown'; const today = new Date().toISOString().slice(0, 10); const key = `rl:${ip}:${today}`; const current = parseInt((await env.RATE_LIMIT.get(key)) || '0', 10); if (current >= DAILY_LIMIT_PER_IP) { return jsonError(origin, 429, '本日の利用上限に達しました。時間をおいて再度お試しください。'); } await env.RATE_LIMIT.put(key, String(current + 1), { expirationTtl: 60 * 60 * 26 }); } // ---- リクエスト内容の検証 ---- let body; try { body = await request.json(); } catch (e) { return jsonError(origin, 400, 'リクエストの形式が正しくありません'); } const { system, messages } = body || {}; if (typeof system !== 'string' || system.length < 10 || system.length > 4000) { return jsonError(origin, 400, '不正なリクエストです'); } if (!Array.isArray(messages) || messages.length === 0 || messages.length > MAX_HISTORY_MESSAGES) { return jsonError(origin, 400, '不正なリクエストです'); } for (const m of messages) { if (!m || (m.role !== 'user' && m.role !== 'assistant') || typeof m.content !== 'string') { return jsonError(origin, 400, '不正なリクエストです'); } } // このWorkerは「会話練習チャット」専用。想定外のsystemプロンプトを使った // 汎用プロキシとして悪用されるのを防ぐための簡易チェック(完全な防御ではありません)。 if (!system.includes('【フィードバックのルール】') || !system.includes('【出力形式】')) { return jsonError(origin, 400, 'このエンドポイントは会話練習チャット専用です'); } // ---- Anthropic APIへ転送 ---- // モデルが説明文やコードフェンス付きでJSONを返してしまい、フロント側で // 解析に失敗する（＝再送信ループになる）のを防ぐため、assistant側の // 返信を "{" から始まるよう強制する（プリフィル）。 const anthropicMessages = messages.concat([{ role: 'assistant', content: '{' }]); let anthropicRes; try { anthropicRes = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: MODEL, max_tokens: MAX_TOKENS, system, messages: anthropicMessages }) }); } catch (e) { return jsonError(origin, 502, 'Anthropic APIへの接続に失敗しました'); } const data = await anthropicRes.json(); // プリフィルで送った "{" は応答に含まれてこないため、ここで先頭に戻しておく。 if (anthropicRes.ok && data && Array.isArray(data.content) && data.content[0] && data.content[0].type === 'text') { data.content[0].text = '{' + data.content[0].text; } return new Response(JSON.stringify(data), { status: anthropicRes.status, headers: { ...corsHeaders(origin), 'Content-Type': 'application/json' } }); } };
